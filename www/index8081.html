Test Commands
# Basic GET 
curl -v http://localhost:8080/

# Upload a file
curl -X POST http://localhost:8080/upload -F "file=@test.txt"

# Delete a file
curl -X DELETE http://localhost:8080/upload/test.txt

# CGI POST
curl -X POST http://localhost:8080/cgi-bin/form.py -d "name=world"

# Trigger 404
curl http://localhost:8080/nonexistent

# Trigger 403
curl http://localhost:8080/errors/403.html

# Trigger 414 (URI too long)
curl "http://localhost:8080/$(python3 -c 'print("a"*10000)')"

# Stress test
siege -c 100 -t 30s http://127.0.0.1:8080/
